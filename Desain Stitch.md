          // Filter card items
          projectItems.forEach(item => {
            const categories = item.getAttribute('data-category') || '';
            if (filter === 'all' || categories.includes(filter)) {
              item.style.display = 'flex';
            } else {
              item.style.display = 'none';
            }
          });
        });
      });
    });
